
async function forceHomepageRails(){
  const userId = user?.id || 1;

  try{
    const res = await fetch("/homepage?user_id="+userId);
    const data = await res.json();

    forceRenderRail("homeContinue", data.continueWatching || [], true);
    forceRenderRail("homeTrending", data.trending || []);
    forceRenderRail("homeAiPicks", data.aiPicks || []);
    forceRenderRail("homeSemantic", data.semantic || []);
    forceRenderRail("homeMovies", data.movies || []);
    forceRenderRail("homeSeries", data.series || []);
  }catch(e){
    console.log("force homepage rails error", e);
  }
}

function forceRenderRail(id, list, isContinue=false){
  const box = document.getElementById(id);
  if(!box) return;

  if(!list || !list.length){
    box.innerHTML = "<span class='small'>Nimic disponibil.</span>";
    return;
  }

  box.innerHTML = list.map(x=>{
    if(isContinue){
      return `
      <div class="card" onclick='resumeContent(${JSON.stringify(x)})'>
        <img src="${x.poster_url || "https://via.placeholder.com/300x450"}">
        <div class="cardInfo">
          <b>${x.title}</b><br>
          <span class="small">Progres: ${x.percent || 0}%</span>
          <div class="progress-bar">
            <div class="progress-fill" style="width:${x.percent || 0}%"></div>
          </div>
          <button onclick='event.stopPropagation();resumeContent(${JSON.stringify(x)})'>Resume</button>
        </div>
      </div>`;
    }

    return card(x);
  }).join("");
}

setTimeout(forceHomepageRails, 800);
setTimeout(forceHomepageRails, 2000);
